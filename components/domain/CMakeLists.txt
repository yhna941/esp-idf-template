set(portable_sources "src/timed_output.cpp" "src/command.cpp" "src/boot_health.cpp")
if(COMMAND idf_component_register)
    idf_component_register(SRCS ${portable_sources}  INCLUDE_DIRS "include")
else()
    add_library(domain STATIC ${portable_sources})
    target_include_directories(domain PUBLIC include)
endif()

    set(component_target ${COMPONENT_LIB})
else()
    set(component_target domain)
endif()
target_compile_features(${component_target} PUBLIC cxx_std_20)
set_target_properties(${component_target} PROPERTIES CXX_EXTENSIONS NO)
