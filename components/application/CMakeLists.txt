set(portable_sources "src/workers.cpp" "src/command_path.cpp" "src/output_quiesce.cpp")
if(COMMAND idf_component_register)
    idf_component_register(SRCS ${portable_sources}  INCLUDE_DIRS "include" REQUIRES domain)
else()
    add_library(application STATIC ${portable_sources})
    target_include_directories(application PUBLIC include)
    target_link_libraries(application PUBLIC domain)
endif()
if(COMMAND idf_component_register)
    set(component_target ${COMPONENT_LIB})
else()
    set(component_target application)
endif()
target_compile_features(${component_target} PUBLIC cxx_std_20)
set_target_properties(${component_target} PROPERTIES CXX_EXTENSIONS NO)
