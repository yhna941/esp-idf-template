set(portable_sources "src/telemetry.cpp" "src/command_envelope.cpp" "src/network_service.cpp")
if(COMMAND idf_component_register)
    idf_component_register(SRCS ${portable_sources} "src/wifi_station.cpp" "src/mqtt_telemetry.cpp" "src/sntp_time.cpp" INCLUDE_DIRS "include" REQUIRES application configuration mqtt esp_event PRIV_REQUIRES esp_netif esp_wifi mbedtls esp_hw_support lwip)
else()
    add_library(networking STATIC ${portable_sources})
    target_include_directories(networking PUBLIC include)
    target_link_libraries(networking PUBLIC application configuration)
endif()
if(COMMAND idf_component_register)
    set(component_target ${COMPONENT_LIB})
else()
    set(component_target networking)
endif()
target_compile_features(${component_target} PUBLIC cxx_std_20)
set_target_properties(${component_target} PROPERTIES CXX_EXTENSIONS NO)
